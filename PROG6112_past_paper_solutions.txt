package productsales;

/** Calculates total, average, max and min sales from a 2D array. */
public class ProductSales implements IProductSales {

    /** Adds every sale in every year/quarter. */
    @Override
    public int TotalSales(int[][] productSales) {
        int total = 0;
        for (int[] year : productSales) {      // each row = one year
            for (int sale : year) {            // each column = one quarter
                total += sale;
            }
        }
        return total;
    }

    /** Total divided by the number of sales figures (cells). */
    @Override
    public double AverageSales(int[][] productSales) {
        int count = 0;
        for (int[] year : productSales) {
            count += year.length;
        }
        return count == 0 ? 0 : (double) TotalSales(productSales) / count;
    }

    /** Starts with the first value, replaces it whenever a bigger one is found. */
    @Override
    public int MaxSale(int[][] productSales) {
        int max = productSales[0][0];
        for (int[] year : productSales) {
            for (int sale : year) {
                if (sale > max) {
                    max = sale;
                }
            }
        }
        return max;
    }

    /** Starts with the first value, replaces it whenever a smaller one is found. */
    @Override
    public int MinSale(int[][] productSales) {
        int min = productSales[0][0];
        for (int[] year : productSales) {
            for (int sale : year) {
                if (sale < min) {
                    min = sale;
                }
            }
        }
        return min;
    }
}

public class ProductSales implements IProductSales {

    private static final int SALES_LIMIT = 500;   // retailer's success limit
    private final int[][] productSales;

    /** Default data from the question (rows = years, columns = products). */
    public ProductSales() {
        this(new int[][]{{300, 150, 700}, {250, 200, 600}});
    }

    /** Lets the unit tests supply their own data. */
    public ProductSales(int[][] productSales) {
        this.productSales = productSales;
    }

    @Override
    public int[][] GetProductSales() {
        return productSales;
    }

    @Override
    public int GetTotalSales() {
        int total = 0;
        for (int[] year : productSales) {
            for (int sale : year) {
                total += sale;
            }
        }
        return total;
    }

    @Override
    public double GetAverageSales() {
        int count = 0;
        for (int[] year : productSales) {
            count += year.length;
        }
        return count == 0 ? 0 : (double) GetTotalSales() / count;
    }

    /** Counts sales above the limit. */
    @Override
    public int GetSalesOverLimit() {
        int count = 0;
        for (int[] year : productSales) {
            for (int sale : year) {
                if (sale > SALES_LIMIT) {
                    count++;
                }
            }
        }
        return count;
    }

    /** Counts sales below the limit. */
    @Override
    public int GetSalesUnderLimit() {
        int count = 0;
        for (int[] year : productSales) {
            for (int sale : year) {
                if (sale < SALES_LIMIT) {
                    count++;
                }
            }
        }
        return count;
    }

    /** Number of years = number of rows in the 2D array. */
    @Override
    public int GetProductsProcessed() {
        return productSales.length;
    }
}

package movietickets;

public class MovieTickets implements IMovieTickets {

    /** Adds up the monthly ticket sales of ONE movie (a single array). */
    @Override
    public int TotalMovieSales(int[] movieTicketSales) {
        int total = 0;
        for (int sales : movieTicketSales) {
            total += sales;
        }
        return total;
    }

    /** Finds the position of the highest total and returns that movie's name. */
    @Override
    public String TopMovie(String[] movies, int[] totalSales) {
        int topIndex = 0;
        for (int i = 1; i < totalSales.length; i++) {
            if (totalSales[i] > totalSales[topIndex]) {
                topIndex = i;
            }
        }
        return movies[topIndex];
    }
}

package movieticketsgui;

public class MovieTickets implements IMovieTickets {

    private static final double VAT_PERCENT = 14.0;

    /** Tickets x price, plus 14% VAT. (3 x 100 -> 342.0) */
    @Override
    public double CalculateTotalTicketPrice(int numberOfTickets, double ticketPrice) {
        return numberOfTickets * ticketPrice * (100 + VAT_PERCENT) / 100.0;
    }

    /** Name cannot be empty; price and number of tickets must be above zero. */
    @Override
    public boolean ValidateData(MovieTicketData movieTicketData) {
        String name = movieTicketData.getMovieName();
        return name != null && !name.trim().isEmpty()
                && movieTicketData.getTicketPrice() > 0
                && movieTicketData.getNumberOfTickets() > 0;
    }
}

package estateagents;

public class EstateAgent implements IEstateAgent {

    private static final double COMMISSION_RATE = 0.02;   // 2%

    /** Total of one agent's monthly sales. */
    @Override
    public double EstateAgentSales(double[] propertySales) {
        double total = 0;
        for (double sale : propertySales) {
            total += sale;
        }
        return total;
    }

    /** 2% of the agent's total sales. */
    @Override
    public double EstateAgentCommission(double propertySales) {
        return propertySales * COMMISSION_RATE;
    }

    /** Returns the INDEX (position) of the agent with the highest total. */
    @Override
    public int TopEstateAgent(double[] totalSales) {
        int topIndex = 0;
        for (int i = 1; i < totalSales.length; i++) {
            if (totalSales[i] > totalSales[topIndex]) {
                topIndex = i;
            }
        }
        return topIndex;
    }
}

package estateagentsgui;

public class EstateAgent implements IEstateAgent {

    /** Commission = price x percentage / 100 (800000 x 10 / 100 = 80000). */
    @Override
    public double CalculateCommission(String propertyPrice, String agentCommission) {
        double price = Double.parseDouble(propertyPrice);
        double percentage = Double.parseDouble(agentCommission);
        return price * percentage / 100;
    }

    /** Location/name not empty; price and percentage above zero. */
    @Override
    public boolean ValidateData(Data dataToValidate) {
        if (isEmpty(dataToValidate.getLocation()) || isEmpty(dataToValidate.getAgentName())) {
            return false;
        }
        try {
            double price = Double.parseDouble(dataToValidate.getPropertyPrice());
            double percentage = Double.parseDouble(dataToValidate.getCommissionPercentage());
            return price > 0 && percentage > 0;
        } catch (NumberFormatException ex) {
            return false;   // not a number = invalid
        }
    }

    private boolean isEmpty(String text) {
        return text == null || text.trim().isEmpty();
    }
}

package estateagentsgui;

import java.awt.BorderLayout;
import java.awt.GridLayout;
import java.io.IOException;
import java.io.PrintWriter;
import javax.swing.JComboBox;
import javax.swing.JFrame;
import javax.swing.JLabel;
import javax.swing.JMenu;

import javax.swing.JOptionPane;
import javax.swing.JPanel;
import javax.swing.JScrollPane;
import javax.swing.JTextArea;
import javax.swing.JTextField;
import javax.swing.SwingUtilities;

public class EstateAgentApp extends JFrame {

    private static final String STARS = "***********************************";

    private final IEstateAgent estateAgent = new EstateAgent();

    private final JComboBox<String> cmbLocation =
            new JComboBox<>(new String[]{"Cape Town", "Durban", "Pretoria"});
    private final JTextField txtName = new JTextField();
    private final JTextField txtPrice = new JTextField();
    private final JTextField txtCommission = new JTextField();
    private final JTextArea txtReport = new JTextArea(7, 25);

    public EstateAgentApp() {
        super("Estate Agent Report");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setSize(420, 460);
        setLocationRelativeTo(null);

        JPanel panel = new JPanel(new GridLayout(5, 2, 5, 10));
        panel.add(new JLabel("AGENT LOCATION:"));
        panel.add(cmbLocation);
        panel.add(new JLabel("ESTATE AGENT NAME:"));
        panel.add(txtName);
        panel.add(new JLabel("PROPERTY PRICE:"));
        panel.add(txtPrice);
        panel.add(new JLabel("COMMISSION PERCENTAGE:"));
        panel.add(txtCommission);
        panel.add(new JLabel("ESTATE AGENT REPORT:"));

        add(panel, BorderLayout.NORTH);
        add(new JScrollPane(txtReport), BorderLayout.CENTER);

        setJMenuBar(buildMenuBar());
    }

    /** File (Exit) and Tools (Process Report, Clear, Save Report). */
    private JMenuBar buildMenuBar() {
        JMenuBar menuBar = new JMenuBar();

        JMenu fileMenu = new JMenu("File");
        JMenuItem miExit = new JMenuItem("Exit");
        miExit.addActionListener(e -> System.exit(0));
        fileMenu.add(miExit);

        JMenu toolsMenu = new JMenu("Tools");
        JMenuItem miProcess = new JMenuItem("Process Report");
        JMenuItem miClear = new JMenuItem("Clear");
        JMenuItem miSave = new JMenuItem("Save Report");
        miProcess.addActionListener(e -> processReport());
        miClear.addActionListener(e -> clearForm());
        miSave.addActionListener(e -> saveReport());
        toolsMenu.add(miProcess);
        toolsMenu.add(miClear);
        toolsMenu.add(miSave);

        menuBar.add(fileMenu);
        menuBar.add(toolsMenu);
        return menuBar;
    }

    /** Validates the input, calculates the commission and shows the report. */
    private void processReport() {
        Data data = new Data((String) cmbLocation.getSelectedItem(),
                txtName.getText(), txtPrice.getText().trim(),
                txtCommission.getText().trim());

        if (!estateAgent.ValidateData(data)) {
            JOptionPane.showMessageDialog(this,
                    "Invalid data: fill in all fields; price and percentage must be above 0.",
                    "Validation", JOptionPane.ERROR_MESSAGE);
            return;
        }

        double commission = estateAgent.CalculateCommission(
                data.getPropertyPrice(), data.getCommissionPercentage());

        txtReport.setText("AGENT LOCATION: " + data.getLocation() + "\n"
                + "AGENT NAME: " + data.getAgentName() + "\n"
                + "PROPERTY PRICE: R " + Double.parseDouble(data.getPropertyPrice()) + "\n"
                + "COMMISSION PERCENTAGE: " + Double.parseDouble(data.getCommissionPercentage()) + "%\n"
                + "CALCULATED COMMISSION: R " + commission);
    }

    /** Saves whatever is in the report area to report.txt. */
    private void saveReport() {
        if (txtReport.getText().isEmpty()) {
            JOptionPane.showMessageDialog(this, "Process the report first.");
            return;
        }
        try (PrintWriter writer = new PrintWriter("report.txt")) {
            writer.println("ESTATE AGENT REPORT");
            writer.println(STARS);
            writer.println(txtReport.getText());
            writer.println(STARS);
            JOptionPane.showMessageDialog(this, "Saved to report.txt");
        } catch (IOException ex) {
            JOptionPane.showMessageDialog(this, "Could not save file: " + ex.getMessage());
        }
    }

    /** Resets all fields and the report area. */
    private void clearForm() {
        cmbLocation.setSelectedIndex(0);
        txtName.setText("");
        txtPrice.setText("");
        txtCommission.setText("");
        txtReport.setText("");
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new EstateAgentApp().setVisible(true));
    }
}
