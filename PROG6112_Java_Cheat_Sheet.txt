2. ARRAYS
  int[] a = {1, 2, 3};                       // 1D
  String[] names = {"A", "B"};
  int[][] s = { {300,150,700}, {250,200,600} };   // 2D: s[row][col]
  s.length          = number of ROWS
  s[0].length       = number of COLUMNS
  for (int[] row : s) { for (int v : row) { ... } }          // for-each
  for (int i = 0; i < s.length; i++) { for (int j = 0; j < s[i].length; j++) { s[i][j] } }

  TOTAL:    int total = 0;  total += v;
  AVERAGE:  (double) total / count          // cast first or you get integer division!
  MAX:      int max = s[0][0];  if (v > max) max = v;
  MIN:      int min = s[0][0];  if (v < min) min = v;
  TOP INDEX (best name/agent/movie):
            int top = 0;
            for (int i = 1; i < totals.length; i++) { if (totals[i] > totals[top]) top = i; }
            return names[top];               // or return top; if the interface returns int
  COUNT over a limit:   if (v > LIMIT) count++;      under:  if (v < LIMIT) count++;
  constant:             private static final int LIMIT = 500;

3. PRINTING / FORMATTING
  System.out.println("text " + value);
  System.out.printf("%-15s", name);        // left-aligned text, width 15
  System.out.printf("%-10d", number);      // integer, width 10
  System.out.printf("%.2f%n", dbl);        // 2 decimals + new line
  String.format("%.0f", 366.67)            // "367"
  Math.round(366.67)                       // 367  (returns long)
  String.format(java.util.Locale.US, "R %,.0f", 4300000.0)   // "R 4,300,000"
  "\n" new line     "\t" tab
  Table: print the heading row first, then one line per row, then totals.

8. SWING GUI SKELETON
  import javax.swing.*;   import java.awt.*;

  public class App extends JFrame {
      private final JTextField txtName = new JTextField();
      private final JComboBox<String> cmb = new JComboBox<>(new String[]{"A","B","C"});
      private final JTextArea txtReport = new JTextArea(6, 25);
      private final JButton btnLoad = new JButton("Load");
      private final JLabel lblInfo = new JLabel("");

      public App() {
          super("Title");
          setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
          setSize(400, 400);
          setLocationRelativeTo(null);                    // centre
          txtReport.setEditable(false);                   // read-only
          JPanel p = new JPanel(new GridLayout(3, 2, 5, 10));   // rows, cols, gaps
          p.add(new JLabel("NAME:"));  p.add(txtName);
          add(p, BorderLayout.NORTH);
          add(new JScrollPane(txtReport), BorderLayout.CENTER);
          setJMenuBar(buildMenuBar());
          btnLoad.addActionListener(e -> doLoad());       // button click
      }
      private JMenuBar buildMenuBar() {
          JMenuBar bar = new JMenuBar();
          JMenu file = new JMenu("File");
          JMenuItem exit = new JMenuItem("Exit");
          exit.addActionListener(e -> System.exit(0));
          file.add(exit);
          JMenu tools = new JMenu("Tools");
          JMenuItem clear = new JMenuItem("Clear");
          clear.addActionListener(e -> doClear());
          tools.add(clear);
          bar.add(file);  bar.add(tools);
          return bar;
      }
      public static void main(String[] args) {
          SwingUtilities.invokeLater(() -> new App().setVisible(true));
      }
  }

  GET:   txt.getText().trim()     cmb.getSelectedItem()     cmb.getSelectedIndex()
  SET:   txt.setText("")          cmb.setSelectedIndex(0)   lbl.setText("...")
  Popup: JOptionPane.showMessageDialog(this, "message", "Title", JOptionPane.ERROR_MESSAGE);
